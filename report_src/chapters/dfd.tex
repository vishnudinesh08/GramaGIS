    import 'package:cloud_firestore/cloud_firestore.dart';
import 'package:flutter/material.dart';
import 'package:quick_bite/classes/manager.dart';
import 'package:quick_bite/classes/managertile.dart';

class ListBranch{
  const ListBranch();

  Widget allbranch(Stream<QuerySnapshot>? stream){
    return StreamBuilder<QuerySnapshot>(
      stream: stream, 
      builder: (context, AsyncSnapshot<QuerySnapshot> snapshot){
        if (snapshot.hasData){
          return Expanded(
            child: ListView.builder(
              physics: const AlwaysScrollableScrollPhysics(),
              shrinkWrap: true,
              itemCount: snapshot.data!.docs.length,
              itemBuilder: (context,index){
                final DocumentSnapshot<Map<String,dynamic>?> snap =
                  snapshot.data!.docs[index] as 
                  DocumentSnapshot<Map<String,dynamic>?>;
                  return Padding(
                    padding: const EdgeInsetsDirectional.only(bottom:
                    15,start: 20,end: 20),
                    child: Card(
                      child: ManagerTile(manager: 
                      Manager.fromFirestore(snap)),
                    ),
                  );
              }
            )
          );
        }else{
          return const Center(child: CircularProgressIndicator(),);
        }
      }
      
    );
  }
}
